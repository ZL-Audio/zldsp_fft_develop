cmake_minimum_required(VERSION 3.20)
project(zldsp_fft LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Build static dependencies")
option(USE_DOUBLE "Use double precision for the implementation under test" OFF)
option(USE_AVX2 "Use AVX2 for benchmarking" OFF)
option(USE_SSE2 "Use the x86-64 SSE2 baseline" OFF)
option(ZLDSP_ENABLE_IPO "Enable IPO for throughput executables" ON)
option(ASAN_TEST "Build the AddressSanitizer test" OFF)
option(STATIC_DISPATCH_EXAMPLE "Build the static dispatch example" OFF)
option(DYNAMIC_DISPATCH_EXAMPLE "Build the caller-owned dispatch example" OFF)
option(BENCHMARK_SELF_TESTS "Build and register correctness checks with CTest" OFF)

if(USE_AVX2 AND USE_SSE2)
    message(FATAL_ERROR "USE_AVX2 and USE_SSE2 cannot both be enabled.")
endif()
set(ZLDSP_ARM FALSE)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64|ARM64")
    set(ZLDSP_ARM TRUE)
    if(USE_AVX2 OR USE_SSE2)
        message(FATAL_ERROR "AVX2/SSE2 cannot be selected for an ARM build.")
    endif()
    if(MSVC)
        set(GLOBAL_ARCH_FLAGS "/arch:armv8.0")
    else()
        set(GLOBAL_ARCH_FLAGS "-march=armv8-a+simd")
    endif()
elseif(MSVC)
    if(USE_AVX2)
        set(GLOBAL_ARCH_FLAGS "/arch:AVX2")
    endif()
else()
    if(USE_AVX2)
        set(GLOBAL_ARCH_FLAGS "-march=x86-64-v3 -maes -mpclmul")
    elseif(USE_SSE2)
        set(GLOBAL_ARCH_FLAGS "-march=x86-64")
    else()
        set(GLOBAL_ARCH_FLAGS "-march=x86-64-v2 -maes -mpclmul")
    endif()
endif()
# Apply the same ISA to dependencies and measured code.
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} ${GLOBAL_ARCH_FLAGS}")
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${GLOBAL_ARCH_FLAGS}")

set(ZLDSP_BENCHMARKS accuracy_cfft accuracy_rfft accuracy_mid_cfft accuracy_mid_rfft
    throughput_cfft throughput_rfft throughput_zldsp_cfft throughput_zldsp_rfft)
set(ZLDSP_BENCHMARK_TARGETS)
set(ZLDSP_THROUGHPUT_TARGETS)
set(ZLDSP_ACCURACY_TARGETS)
foreach(name IN LISTS ZLDSP_BENCHMARKS)
    string(TOUPPER "${name}_TEST" flag)
    option(${flag} "Build ${name}" OFF)
    if(${flag})
        set(target zldsp_fft_${name})
        string(REGEX MATCH "^[^_]+" runner "${name}")
        add_executable(${target} benchmark/${runner}.cpp)
        if(name MATCHES "_rfft$")
            target_compile_definitions(${target} PRIVATE ZLB_REAL_FFT)
        endif()
        if(name MATCHES "^accuracy_(cfft|rfft)$|^throughput_zldsp_")
            target_compile_definitions(${target} PRIVATE ZLB_ALL_APIS)
        endif()
        list(APPEND ZLDSP_BENCHMARK_TARGETS ${target})
        if(name MATCHES "^throughput")
            list(APPEND ZLDSP_THROUGHPUT_TARGETS ${target})
        else()
            list(APPEND ZLDSP_ACCURACY_TARGETS ${target})
        endif()
    endif()
endforeach()

set(ZLDSP_LIBRARY_ACCURACY_TARGETS)
foreach(kind cfft rfft)
    if(TARGET zldsp_fft_accuracy_mid_${kind})
        list(APPEND ZLDSP_LIBRARY_ACCURACY_TARGETS zldsp_fft_accuracy_mid_${kind})
    endif()
endforeach()

if(ZLDSP_THROUGHPUT_TARGETS)
    set(BENCHMARK_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
    set(BENCHMARK_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
    set(BENCHMARK_DOWNLOAD_DEPENDENCIES OFF CACHE BOOL "" FORCE)
    add_subdirectory(google/benchmark EXCLUDE_FROM_ALL)
    foreach(target IN LISTS ZLDSP_THROUGHPUT_TARGETS)
        target_link_libraries(${target} PRIVATE benchmark::benchmark)
    endforeach()
endif()

set(ZLDSP_BACKENDS ZLDSP PFFFT VDSP IPP ARMPL ARMPL_ESTIMATE KFR FFTW3 FFTW3_ESTIMATE)
set(selected_backends)
foreach(backend IN LISTS ZLDSP_BACKENDS)
    option(ENABLE_${backend} "Select ${backend} as the benchmark implementation" OFF)
    if(ENABLE_${backend})
        list(APPEND selected_backends ${backend})
    endif()
endforeach()
list(LENGTH selected_backends backend_count)
if(backend_count GREATER 1)
    message(FATAL_ERROR "Select only one ENABLE_<library> backend per build directory.")
elseif(backend_count EQUAL 0)
    set(ENABLE_ZLDSP ON)
    set(selected_backends ZLDSP)
endif()
add_library(zldsp_benchmark_backend INTERFACE)
target_compile_definitions(zldsp_benchmark_backend INTERFACE ENABLE_${selected_backends})
foreach(target IN LISTS ZLDSP_BENCHMARK_TARGETS)
    target_link_libraries(${target} PRIVATE zldsp_benchmark_backend)
    if(USE_DOUBLE)
        target_compile_definitions(${target} PRIVATE USE_DOUBLE)
    endif()
endforeach()

if(ENABLE_IPP)
    if(ZLDSP_ARM)
        message(FATAL_ERROR "Intel IPP requires an x86 build.")
    endif()
    if(WIN32)
        set(default_ipp_root "$ENV{USERPROFILE}/.nuget/packages/intelipp.static.win-x64/2022.3.0.387")
    elseif(APPLE)
        set(default_ipp_root "$ENV{HOME}/.nuget/packages/intelipp.mac/2021.2.0/static/lib/native")
    else()
        set(default_ipp_root "$ENV{HOME}/.nuget/packages/intelipp.static.linux-x64/2021.2.0.233")
    endif()
    if(DEFINED ENV{IPPROOT})
        set(default_ipp_root "$ENV{IPPROOT}")
    endif()
    if(DEFINED ENV{IPP_ROOT})
        set(default_ipp_root "$ENV{IPP_ROOT}")
    endif()
    set(IPP_ROOT "${default_ipp_root}" CACHE PATH "Intel IPP installation root")
    find_path(IPP_INCLUDE_DIR ipps.h HINTS "${IPP_ROOT}"
        PATH_SUFFIXES include include/ipp lib/native/include build/native/include/ipp)
    set(ipp_libraries)
    foreach(component ipps ippcore ippvm)
        find_library(IPP_${component}_LIBRARY NAMES ${component}mt ${component} HINTS "${IPP_ROOT}"
            PATH_SUFFIXES lib lib/intel64 lib/intel64_lin lib/intel64_win lib/native/linux-x64 osx-x64 build/native/win-x64)
        list(APPEND ipp_libraries ${IPP_${component}_LIBRARY})
    endforeach()
    if(IPP_INCLUDE_DIR AND IPP_ipps_LIBRARY AND IPP_ippcore_LIBRARY AND IPP_ippvm_LIBRARY)
        add_library(zldsp_ipp INTERFACE)
        target_include_directories(zldsp_ipp INTERFACE "${IPP_INCLUDE_DIR}")
        target_link_libraries(zldsp_ipp INTERFACE ${ipp_libraries})
        if(USE_AVX2)
            target_compile_definitions(zldsp_ipp INTERFACE IPP_USE_AVX2)
        endif()
        target_link_libraries(zldsp_benchmark_backend INTERFACE zldsp_ipp)
    else()
        message(FATAL_ERROR "Intel IPP headers/libraries not found. Set IPP_ROOT.")
    endif()
endif()

if (ENABLE_ARMPL OR ENABLE_ARMPL_ESTIMATE)
    if(ZLDSP_LIBRARY_ACCURACY_TARGETS)
        message(FATAL_ERROR "Build ArmPL throughput separately: library accuracy links FFTW's conflicting API symbols.")
    endif()
    set(ARMPL_DIR "$ENV{ARMPL_DIR}" CACHE PATH "Arm Performance Libraries installation directory")

    find_path(ARMPL_INCLUDE_DIR
        NAMES armpl.h
        HINTS "${ARMPL_DIR}"
        PATH_SUFFIXES include)
    find_path(ARMPL_FFTW_INCLUDE_DIR
        NAMES fftw3.h
        HINTS "${ARMPL_DIR}"
        PATH_SUFFIXES include)
    find_library(ARMPL_LIBRARY
        NAMES armpl_lp64 armpl
        HINTS "${ARMPL_DIR}"
        PATH_SUFFIXES lib)

    if (NOT ARMPL_INCLUDE_DIR OR NOT ARMPL_FFTW_INCLUDE_DIR OR NOT ARMPL_LIBRARY)
        message(FATAL_ERROR "Arm Performance Libraries headers or library NOT FOUND. Set ARMPL_DIR to the installation directory.")
    endif ()

    set(ARMPL_RUNTIME_LIBRARIES)
    if (APPLE)
        find_library(ARMPL_FORTRAN_RUNTIME NAMES FortranRuntime HINTS "${ARMPL_DIR}" PATH_SUFFIXES lib)
        find_library(ARMPL_FORTRAN_DECIMAL NAMES FortranDecimal HINTS "${ARMPL_DIR}" PATH_SUFFIXES lib)
        if (ARMPL_FORTRAN_RUNTIME)
            list(APPEND ARMPL_RUNTIME_LIBRARIES ${ARMPL_FORTRAN_RUNTIME})
        endif ()
        if (ARMPL_FORTRAN_DECIMAL)
            list(APPEND ARMPL_RUNTIME_LIBRARIES ${ARMPL_FORTRAN_DECIMAL})
        endif ()
    elseif (UNIX)
        find_library(ARMPL_FORTRAN_RUNTIME NAMES gfortran HINTS "${ARMPL_DIR}" PATH_SUFFIXES lib)
        if (ARMPL_FORTRAN_RUNTIME)
            list(APPEND ARMPL_RUNTIME_LIBRARIES ${ARMPL_FORTRAN_RUNTIME})
        endif ()
    endif ()

    message(STATUS "Arm Performance Libraries found at ${ARMPL_DIR}")
    target_include_directories(zldsp_benchmark_backend INTERFACE "${ARMPL_INCLUDE_DIR}" "${ARMPL_FFTW_INCLUDE_DIR}")
    target_link_libraries(zldsp_benchmark_backend INTERFACE ${ARMPL_LIBRARY} ${ARMPL_RUNTIME_LIBRARIES})
endif ()

if(ENABLE_KFR OR ZLDSP_LIBRARY_ACCURACY_TARGETS)
    set(KFR_USE_BOOST_MATH OFF CACHE BOOL "" FORCE)
    set(KFR_USE_BOOST OFF CACHE BOOL "" FORCE)
    set(KFR_ENABLE_MULTIARCH OFF CACHE BOOL "" FORCE)
    set(KFR_ENABLE_AUDIO OFF CACHE BOOL "" FORCE)
    set(KFR_ENABLE_IO OFF CACHE BOOL "" FORCE)
    set(KFR_ENABLE_DSP OFF CACHE BOOL "" FORCE)
    if(NOT ZLDSP_ARM)
        if(USE_AVX2)
            set(KFR_ARCH avx2 CACHE STRING "" FORCE)
        elseif(USE_SSE2)
            set(KFR_ARCH sse2 CACHE STRING "" FORCE)
        else()
            set(KFR_ARCH sse42 CACHE STRING "" FORCE)
        endif()
    endif()
    add_subdirectory(kfr_impl/kfr EXCLUDE_FROM_ALL)
    if(ENABLE_KFR)
        target_link_libraries(zldsp_benchmark_backend INTERFACE kfr kfr_dft)
    endif()
endif()
if(ENABLE_FFTW3 OR ENABLE_FFTW3_ESTIMATE OR ZLDSP_LIBRARY_ACCURACY_TARGETS)
    set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
    if(NOT USE_DOUBLE)
        set(ENABLE_FLOAT ON CACHE BOOL "" FORCE)
        set(FFTW_TARGET fftw3f)
    else()
        set(ENABLE_FLOAT OFF CACHE BOOL "" FORCE)
        set(FFTW_TARGET fftw3)
    endif()
    set(ENABLE_AVX OFF CACHE BOOL "" FORCE)
    set(ENABLE_AVX2 ${USE_AVX2} CACHE BOOL "" FORCE)
    set(ENABLE_NEON ${ZLDSP_ARM} CACHE BOOL "" FORCE)
    if(NOT ZLDSP_ARM AND NOT USE_AVX2)
        set(ENABLE_SSE2 ON CACHE BOOL "" FORCE)
    else()
        set(ENABLE_SSE2 OFF CACHE BOOL "" FORCE)
    endif()
    add_subdirectory(fftw3_impl/fftw3 EXCLUDE_FROM_ALL)
    get_target_property(fftw_defs ${FFTW_TARGET} COMPILE_DEFINITIONS)
    if(fftw_defs)
        list(REMOVE_ITEM fftw_defs "/bigobj")
        set_target_properties(${FFTW_TARGET} PROPERTIES COMPILE_DEFINITIONS "${fftw_defs}")
    endif()
    if(MSVC)
        target_compile_options(${FFTW_TARGET} PRIVATE /bigobj)
    endif()
    if(ENABLE_FFTW3 OR ENABLE_FFTW3_ESTIMATE)
        target_link_libraries(zldsp_benchmark_backend INTERFACE ${FFTW_TARGET})
        target_include_directories(zldsp_benchmark_backend INTERFACE fftw3_impl/fftw3/api)
    endif()
endif()
foreach(target IN LISTS ZLDSP_LIBRARY_ACCURACY_TARGETS)
    target_link_libraries(${target} PRIVATE kfr kfr_dft ${FFTW_TARGET})
    target_include_directories(${target} PRIVATE fftw3_impl/fftw3/api)
endforeach()
